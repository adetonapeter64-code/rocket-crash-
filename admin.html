Tap below to play." ); return; } if (text === "/stop") { user.notificationsEnabled = false; saveDB(); await telegram("sendMessage", { chat_id: msg.chat.id, text: "🔕 Daily notifications are now OFF.\n\n" + "Send /start anytime to turn them back ON." }); return; } if (text === "/help") { await telegram("sendMessage", { chat_id: msg.chat.id, text: "🎮 Commands\n\n" + "/start — open the game\n" + "/game — play now\n" + "/stop — stop daily notifications\n" + "/help — show this help" }); return; } await sendGameMessage( msg.chat.id, "🎮 The game is ready. } } } function lagosParts() { const parts = new Intl.DateTimeFormat( "en-GB", { timeZone: TIMEZONE, year: "numeric", month: "2-digit", day: "2-digit", hour: "2-digit", minute: "2-digit", hour12: false } ).formatToParts(new Date());
